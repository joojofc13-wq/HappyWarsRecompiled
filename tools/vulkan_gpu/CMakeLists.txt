cmake_minimum_required(VERSION 3.25)
project(happywars_vulkan_plugin LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
find_package(rexglue CONFIG REQUIRED)
set(REXGLUE_USE_D3D12 OFF)
set(REXGLUE_USE_VULKAN ON)
set(REXGLUE_ROOT "" CACHE PATH "Pinned ReXGlue source")
get_target_property(runtime_definitions rex::runtime INTERFACE_COMPILE_DEFINITIONS)
list(FILTER runtime_definitions EXCLUDE REGEX "^REX_HAS_D3D12=")
set_property(TARGET rex::runtime PROPERTY INTERFACE_COMPILE_DEFINITIONS "${runtime_definitions}")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(VULKAN_HEADERS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(VULKAN_HEADERS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(VULKAN_HEADERS_ENABLE_MODULE OFF CACHE BOOL "" FORCE)
add_subdirectory(${REXGLUE_ROOT}/thirdparty/vulkan-headers vulkan-headers)
set(VMA_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(${REXGLUE_ROOT}/thirdparty/vulkan-memory-allocator vulkan-memory-allocator)
add_library(spirv-tools-headers INTERFACE)
target_include_directories(spirv-tools-headers SYSTEM INTERFACE ${REXGLUE_ROOT}/thirdparty/spirv-tools/include)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_RTTI ON CACHE BOOL "" FORCE)
set(SKIP_GLSLANG_INSTALL ON CACHE BOOL "" FORCE)
add_compile_options(-fno-strict-aliasing -ffp-model=strict -fno-char8_t)
add_subdirectory(${REXGLUE_ROOT}/thirdparty/glslang glslang)
if(NOT TARGET glslang::SPIRV)
  add_library(glslang::SPIRV ALIAS SPIRV)
endif()
function(rex_target_pch target)
 target_precompile_headers(${target} PRIVATE ${ARGN})
endfunction()
function(rexglue_add_version_resource target)
endfunction()
add_library(rexruntime ALIAS rex::runtime)
add_library(xxHash::xxhash ALIAS rex::xxhash)
include(${CMAKE_CURRENT_SOURCE_DIR}/gpu-target.cmake)
file(GLOB VULKAN_UI_SOURCES CONFIGURE_DEPENDS ${REXGLUE_ROOT}/src/ui/vulkan/*.cpp)
target_sources(rexgpu-xenos PRIVATE ${VULKAN_UI_SOURCES})
target_compile_definitions(rexgpu-xenos PRIVATE REX_HAS_VULKAN=1 REX_HAS_D3D12=0)
target_include_directories(rexgpu-xenos PRIVATE ${REXGLUE_ROOT}/src)
target_link_libraries(rexgpu-xenos PRIVATE spirv-tools-headers)
rexglue_apply_target_settings(rexgpu-xenos)
